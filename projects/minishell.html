🐚 MiniShell

A lightweight Unix command-line interpreter written in C that replicates core Linux shell functionality — built-in commands, external process execution, pipe support, background jobs, signal handling, and a custom prompt — all built from scratch using Linux system calls.

Project Overview

MiniShell is a fully functional shell implemented in C using Linux system calls. It reads user input in a REPL loop, classifies each command as built-in, external, or unknown, and dispatches accordingly. External commands are executed via fork() + execvp(). Pipe chains between multiple commands are handled using pipe() and dup2(). Background process execution is supported with &. Signal handling for SIGINT, SIGTSTP, and SIGCHLD enables foreground/background job control. Suspended jobs are tracked using a singly linked list and can be resumed with fg or bg.

Core Features

Built-in Commands

Implements cd, pwd, echo, exit, jobs, fg, bg and more directly inside the shell process.

External Commands

Executes any system binary using fork() + execvp(). Scans PATH environment and ext_cmd.txt for command resolution.

Pipe Support

Chains multiple commands with | using pipe() and dup2() to wire stdout of one process to stdin of the next.

Background Jobs

Append & to run a process in the background. Job PIDs are stored in a linked list and can be brought back with fg/bg.

Signal Handling

Handles SIGINT (Ctrl+C), SIGTSTP (Ctrl+Z), and SIGCHLD for automatic zombie cleanup and job tracking.

Custom Prompt

Supports PS1= to change the prompt at runtime, just like a real bash shell.

/* ── Signal handler ─────────────────────────────────────────────────────── */
void signal_handler(int signum) {

    if (signum == SIGINT) {
        if (pid > 0) {
            kill(pid, SIGINT);
            pid = -1;
            printf("\n");
        } else {
            printf("\n\x1b[32m%s\x1b[0m", prompt);
            fflush(stdout);
        }
    }
    else if (signum == SIGTSTP) {
        if (pid > 0) {
            /* Track suspended PID in linked list if not already there */
            SLL *temp  = list;
            int  found = 0;
            while (temp != NULL) {
                if (temp->data == pid) { found = 1; break; }
                temp = temp->link;
            }
            if (!found) insert_at_first(&list, pid);
            printf("\n");
        } else {
            printf("\n\x1b[32m%s\x1b[0m", prompt);
            fflush(stdout);
        }
    }
    else if (signum == SIGCHLD) {
        /* Non-blocking wait — reap zombie children */
        int   child_status;
        pid_t child_pid = waitpid(-1, &child_status, WNOHANG);
        if (child_pid > 0) {
            status = child_status;
            if (child_pid == pid) pid = -1;
            /* Remove completed PID from job list */
            SLL *temp = list, *prev = NULL;
            while (temp != NULL) {
                if (temp->data == child_pid) {
                    if (prev == NULL) delete_at_first(&list);
                    else { prev->link = temp->link; free(temp); }
                    break;
                }
                prev = temp; temp = temp->link;
            }
        }
    }
}

/* ── Main REPL loop ─────────────────────────────────────────────────────── */
void scan_input(char *prompt, char *input_string) {
    signal(SIGINT,  signal_handler);
    signal(SIGTSTP, signal_handler);
    signal(SIGCHLD, signal_handler);
    extract_external_commands(external_commands);

    while (1) {
        printf("\x1b[32m%s\x1b[0m", prompt);
        input_string[0] = '\0';
        scanf("%[^\n]", input_string);
        getchar();

        if (strlen(input_string) == 0) continue;

        /* PS1= — custom prompt */
        if (strncmp(input_string, "PS1=", 4) == 0) {
            if (strlen(input_string) == 4)       prompt[0] = '\0';
            else if (!strcmp(input_string + 4, " ")) printf("Error: command not found\n");
            else                                  strcpy(prompt, input_string + 4);
            continue;
        }

        char *cmd  = get_command(input_string);
        if (!cmd || strlen(cmd) == 0) { free(cmd); continue; }

        int type = check_command_type(cmd);

        if (type == BUILTIN)   { execute_internal_commands(input_string); status = 0; }
        else if (type == EXTERNAL) { execute_external_commands(input_string); }
        else                  { printf("%s: Command not found\n", cmd); status = 127 << 8; }

        free(cmd);
    }
}

/* ── Extract first token from input ─────────────────────────────────────── */
char *get_command(char *input_string) {
    if (!input_string || strlen(input_string) == 0) return NULL;

    int start = 0;
    while (input_string[start] == ' ') start++;

    int i = start;
    while (input_string[i] != ' ' && input_string[i] != '\0') i++;

    if (i == start) return NULL;

    char *cmd = malloc(i - start + 1);
    if (!cmd) return NULL;
    strncpy(cmd, input_string + start, i - start);
    cmd[i - start] = '\0';
    return cmd;
}

/* ── Determine if command is BUILTIN, EXTERNAL, or unknown ──────────────── */
int check_command_type(char *command) {
    if (!command || strlen(command) == 0) return NO_COMMAND;

    /* Check built-ins list */
    for (int i = 0; builtins[i] != NULL; i++)
        if (strcmp(command, builtins[i]) == 0) return BUILTIN;

    /* Check ext_cmd.txt list */
    if (external_commands[0] != NULL)
        for (int i = 0; external_commands[i] != NULL; i++)
            if (strcmp(command, external_commands[i]) == 0) return EXTERNAL;

    /* Fallback: search PATH */
    char *path = getenv("PATH");
    if (path) {
        char *path_copy = strdup(path);
        char *dir       = strtok(path_copy, ":");
        char  full_path[1024];
        while (dir) {
            snprintf(full_path, sizeof(full_path), "%s/%s", dir, command);
            if (access(full_path, X_OK) == 0) { free(path_copy); return EXTERNAL; }
            dir = strtok(NULL, ":");
        }
        free(path_copy);
    }
    return NO_COMMAND;
}

/* ── Load external command list from ext_cmd.txt ────────────────────────── */
void extract_external_commands(char **external_commands) {
    FILE *fd = fopen("./ext_cmd.txt", "r");
    if (!fd) { perror("fopen"); external_commands[0] = NULL; return; }

    char buffer[256];
    int  size = 0;
    while (fgets(buffer, sizeof(buffer), fd) && size < 151) {
        int len = strlen(buffer);
        if (len > 0 && (buffer[len-1] == '\n' || buffer[len-1] == '\r')) buffer[--len] = '\0';
        external_commands[size] = malloc(len + 1);
        if (!external_commands[size]) { fclose(fd); external_commands[size] = NULL; return; }
        strcpy(external_commands[size++], buffer);
    }
    external_commands[size] = NULL;
    fclose(fd);
}

/* ── Execute built-in commands ──────────────────────────────────────────── */
void execute_internal_commands(char *input_string) {

    if (!strcmp(input_string, "exit"))  exit(0);

    else if (!strcmp(input_string, "pwd")) {
        char buf[256];
        if (getcwd(buf, 256)) printf("%s\n", buf);
        else perror("getcwd");
    }
    else if (!strncmp(input_string, "cd", 2)) {
        char buf[256], *path = NULL;
        if      (strlen(input_string) == 2)  path = getenv("HOME");
        else if (input_string[2] == ' ')   { path = input_string + 3; while (*path == ' ') path++; }
        else                                  { printf("cd: invalid syntax\n"); return; }
        if (chdir(path) == -1) fprintf(stderr, "cd: %s: No such file or directory\n", path);
        else if (getcwd(buf, 256))           printf("%s\n", buf);
    }
    else if (!strcmp(input_string, "echo $$"))     printf("%d\n", getpid());
    else if (!strcmp(input_string, "echo $?"))
        printf("%d\n", WIFEXITED(status) ? WEXITSTATUS(status) : 0);
    else if (!strcmp(input_string, "echo $SHELL")) {
        char *shell = getenv("SHELL");
        printf("%s\n", shell ? shell : "SHELL not set");
    }
    else if (!strcmp(input_string, "jobs")) print_data(list);
    else if (!strcmp(input_string, "fg")) {
        if (list) {
            pid_t job_pid = list->data;
            pid = job_pid;
            printf("[%d] continued\n", job_pid);
            kill(job_pid, SIGCONT);
            waitpid(job_pid, &status, WUNTRACED);
            if (WIFEXITED(status) || WIFSIGNALED(status)) delete_at_first(&list);
            pid = -1;
        } else printf("fg: no current job\n");
    }
    else if (!strcmp(input_string, "bg")) {
        if (list) {
            pid_t bg_pid = list->data;
            printf("[%d] continued\n", bg_pid);
            kill(bg_pid, SIGCONT);
            delete_at_first(&list);
        } else printf("bg: no current job\n");
    }
}

/* ── Execute external commands (with pipe support) ──────────────────────── */
void execute_external_commands(char *input_string) {
    char *commands[100];
    int   arg_count = 0, background = 0;
    char  input[256];
    strcpy(input, input_string);

    /* Detect trailing & for background execution */
    int len = strlen(input);
    for (int i = len - 1; i >= 0; i--) {
        if      (input[i] == '&') { background = 1; input[i] = ' '; break; }
        else if (input[i] != ' ') break;
    }

    /* Tokenise input */
    char *token = strtok(input, " ");
    while (token && arg_count < 99) { commands[arg_count++] = token; token = strtok(NULL, " "); }
    commands[arg_count] = NULL;

    /* Check for pipe character */
    int found = 0;
    for (int i = 0; i < arg_count; i++)
        if (!strcmp(commands[i], "|")) { found = 1; break; }

    if (!found) {
        /* Simple single command */
        pid = fork();
        if (pid < 0)       { perror("fork failed"); return; }
        else if (pid == 0) {
            signal(SIGINT,  SIG_DFL);
            signal(SIGTSTP, SIG_DFL);
            signal(SIGCHLD, SIG_DFL);
            execvp(commands[0], commands);
            perror("exec failed"); exit(1);
        } else {
            if (background) { printf("[%d]\n", pid); insert_at_first(&list, pid); pid = -1; }
            else            { waitpid(pid, &status, WUNTRACED); if (!WIFSTOPPED(status)) pid = -1; }
        }
    } else {
        /* Pipe chain — split commands by | and wire file descriptors */
        int arr[arg_count], count = 0, pipe_count = 0;
        arr[count++] = 0;
        for (int i = 0; i < arg_count; i++)
            if (!strcmp(commands[i], "|")) { commands[i] = NULL; arr[count++] = i + 1; pipe_count++; }

        int fd1[2], fd2[2];
        for (int i = 0; i <= pipe_count; i++) {
            if (i < pipe_count) { if (pipe(fd2) == -1) { perror("pipe"); exit(1); } }
            pid = fork();
            if (pid == 0) {
                if (i > 0)           { dup2(fd1[0], STDIN_FILENO);  close(fd1[0]); close(fd1[1]); }
                if (i < pipe_count) { dup2(fd2[1], STDOUT_FILENO); close(fd2[0]); close(fd2[1]); }
                execvp(commands[arr[i]], commands + arr[i]); exit(1);
            } else if (pid > 0) {
                if (i > 0) { close(fd1[0]); close(fd1[1]); }
                fd1[0] = fd2[0]; fd1[1] = fd2[1];
            } else { perror("fork"); exit(1); }
        }
        if (pipe_count > 0) { close(fd1[0]); close(fd1[1]); }
        for (int i = 0; i <= pipe_count; i++) wait(&status);
        pid = -1;
    }
}

/* ── Insert PID at head of job list ─────────────────────────────────────── */
int insert_at_first(SLL **head, int data) {
    SLL *new = malloc(sizeof(SLL));
    if (!new) return 0;
    new->data = data;
    new->link = NULL;
    if (*head == NULL) { *head = new; return 1; }
    new->link = *head;
    *head = new;
    return 1;
}

/* ── Remove head node from job list ─────────────────────────────────────── */
int delete_at_first(SLL **head) {
    if (*head == NULL) return 0;
    SLL *temp = *head;
    *head = temp->link;
    free(temp);
    return 1;
}

/* ── Print all tracked job PIDs ─────────────────────────────────────────── */
void print_data(SLL *head) {
    if (!head) { printf("No Jobs\n"); return; }
    SLL *temp = head;
    while (temp) {
        printf("%d ", temp->data);
        temp = temp->link;
    }
    printf("\n");
}

Project Architecture

Key Challenges & Solutions

Signal Safety

Distinguishing Ctrl+C and Ctrl+Z behaviour between shell-level and child-process-level required careful use of pid state and re-raising signals correctly.

Pipe Chaining

Generalised pipe support for N commands required iterating over | tokens and wiring fd1/fd2 pairs correctly between consecutive forks.

Zombie Reaping

Used SIGCHLD with waitpid(WNOHANG) to non-blockingly reap all finished children and keep the job list consistent.
